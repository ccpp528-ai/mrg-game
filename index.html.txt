        <div style="color: var(--main-gold); letter-spacing: 5px;">GOLD RESERVE</div>
    </div>

    <div class="coin-section">
        <div class="mrg-coin" id="coin"></div>
    </div>

    <div class="energy-box">
        <div class="progress-bar"><div class="progress-fill" id="fill"></div></div>
        <div style="margin-top: 10px; display: flex; justify-content: space-between;">
            <span>⚡ Energy</span>
            <span id="energyText">1000/1000</span>
        </div>
    </div>

    <script>
        let tg = window.Telegram.WebApp;
        tg.expand(); // لفتح التطبيق بكامل الشاشة

        let score = parseInt(localStorage.getItem('mrg_score')) || 0;
        let energy = 1000;

        function update() {
            document.getElementById('score').innerText = score.toLocaleString();
            document.getElementById('fill').style.width = (energy/10) + "%";
            document.getElementById('energyText').innerText = energy + "/1000";
        }

        document.getElementById('coin').addEventListener('pointerdown', (e) => {
            if(energy > 0) {
                score += 1; energy -= 1;
                createFloat(e.clientX, e.clientY);
                localStorage.setItem('mrg_score', score);
                update();
                if(tg.HapticFeedback) tg.HapticFeedback.impactOccurred('medium');
            }
        });

        function createFloat(x, y) {
            let div = document.createElement('div');
            div.className = 'floating-text';
            div.innerText = "+1";
            div.style.left = x + "px";
            div.style.top = y + "px";
            document.body.appendChild(div);
            setTimeout(() => div.remove(), 700);
        }

        setInterval(() => { if(energy < 1000) { energy += 1; update(); } }, 2000);
        update();
    </script>
